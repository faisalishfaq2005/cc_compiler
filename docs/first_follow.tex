\section{Introduction}
% ------------------------------------------------------------------

FIRST and FOLLOW sets are the mathematical foundation of LL(1) parsing.
They are computed once from the grammar productions and serve three roles:

\begin{enumerate}
  \item \textbf{LL(1) table construction}: entry $M[A, a]$ is set to the
        production $A \to \alpha$ whenever $a \in \text{FIRST}(\alpha)$,
        or whenever $\varepsilon \in \text{FIRST}(\alpha)$ and
        $a \in \text{FOLLOW}(A)$.
  \item \textbf{Panic-mode error recovery}: after a syntax error, the parser
        discards input tokens until it finds a token in the FOLLOW set of the
        current non-terminal.
  \item \textbf{SLR(1) reduce actions}: the parser reduces by $A \to \alpha$
        in state $s$ only if the next input symbol belongs to $\text{FOLLOW}(A)$.
\end{enumerate}

% ------------------------------------------------------------------
\section{Algorithm: FIRST Sets}
% ------------------------------------------------------------------

\begin{lstlisting}[style=pseudo]
for each terminal $a$:   FIRST($a$) := {$a$}
for each non-terminal $A$: FIRST($A$) := $\emptyset$

repeat until no FIRST set changes:
  for each production $A \to X_1 X_2 \cdots X_k$:
    add FIRST($X_1$) \ {$\varepsilon$}  to  FIRST($A$)
    for $i$ = 1 to $k-1$:
      if $\varepsilon \in$ FIRST($X_1$) $\cap \cdots \cap$ FIRST($X_i$):
        add FIRST($X_{i+1}$) \ {$\varepsilon$}  to  FIRST($A$)
    if $\varepsilon \in$ FIRST($X_1$) $\cap \cdots \cap$ FIRST($X_k$):
      add $\varepsilon$  to  FIRST($A$)
  for each production $A \to \varepsilon$:
    add $\varepsilon$  to  FIRST($A$)
\end{lstlisting}

\subsection{Worked Example: FIRST(AddExpr)}

\begin{align*}
  \textit{AddExpr} &\to \textit{MulExpr}\;\textit{AddTail}\\
  \text{FIRST}(\textit{AddExpr}) &= \text{FIRST}(\textit{MulExpr}) \setminus \{\varepsilon\}
    \quad (\textit{MulExpr} \text{ cannot derive } \varepsilon)\\
  \textit{MulExpr} &\to \textit{UnaryExpr}\;\textit{MulTail}\\
  \textit{UnaryExpr} &\to \texttt{!}\;\textit{UnaryExpr}
                        \mid \texttt{-}\;\textit{UnaryExpr}
                        \mid \textit{PostExpr}\\
  \textit{PostExpr} &\to \textit{Primary}\;\textit{PostTail}\\
  \text{FIRST}(\textit{Primary}) &= \{
    \texttt{(},\; \text{BOOL\_CONST},\; \text{DOUBLE\_CONST},\; \text{INT\_CONST},\\
    &\quad \texttt{New},\; \texttt{NewArray},\; \texttt{ReadInteger},\; \texttt{ReadLine},\\
    &\quad \text{STRING\_CONST},\; \texttt{id},\; \texttt{null},\; \texttt{this}
  \}
\end{align*}

Propagating upward:
\[
  \text{FIRST}(\textit{AddExpr}) = \{\texttt{!},\; \texttt{-}\}
    \cup \text{FIRST}(\textit{Primary})
\]

% ------------------------------------------------------------------
\section{Algorithm: FOLLOW Sets}
% ------------------------------------------------------------------

\begin{lstlisting}[style=pseudo]
FOLLOW(Start) := {$}
for all other non-terminals $A$: FOLLOW($A$) := $\emptyset$

repeat until no FOLLOW set changes:
  for each production $B \to \alpha\,A\,\beta$:
    add FIRST($\beta$) \ {$\varepsilon$}   to   FOLLOW($A$)
    if $\varepsilon \in$ FIRST($\beta$):
      add FOLLOW($B$)   to   FOLLOW($A$)
  for each production $B \to \alpha\,A$:
    add FOLLOW($B$)   to   FOLLOW($A$)
\end{lstlisting}

\subsection{Worked Example: FOLLOW(OrExpr)}

From the production \textit{Expr} $\to$ \textit{OrExpr} \textit{AssignTail}:
\begin{align*}
  \text{FIRST}(\textit{AssignTail}) &= \{\texttt{=},\, \varepsilon\}\\
  \Rightarrow\quad \text{FOLLOW}(\textit{OrExpr}) &\supseteq \{\texttt{=}\}\\
  \varepsilon \in \text{FIRST}(\textit{AssignTail}) &\Rightarrow
    \text{FOLLOW}(\textit{OrExpr}) \supseteq \text{FOLLOW}(\textit{Expr})
    = \{\texttt{)},\texttt{,},\texttt{;},\texttt{]}\}\\
  \Rightarrow\quad \text{FOLLOW}(\textit{OrExpr}) &= \{=, ), {,}, {;}, ]\}
\end{align*}
